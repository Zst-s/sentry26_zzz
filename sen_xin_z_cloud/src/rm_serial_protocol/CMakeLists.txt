cmake_minimum_required(VERSION 3.8)
project(rm_serial_protocol)
find_package(ament_cmake REQUIRED)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(protocol_layout_test test/protocol_layout_test.cpp)
  target_include_directories(protocol_layout_test PRIVATE include)
  target_compile_features(protocol_layout_test PRIVATE cxx_std_17)
  add_test(NAME protocol_layout COMMAND protocol_layout_test)
endif()
ament_package()

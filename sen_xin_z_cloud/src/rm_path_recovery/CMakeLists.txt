cmake_minimum_required(VERSION 3.8)
project(rm_path_recovery)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
set(deps nav2_behaviors nav2_behavior_tree nav2_core nav2_costmap_2d nav2_util
  nav2_msgs nav_msgs geometry_msgs builtin_interfaces rclcpp rclcpp_action
  rclcpp_lifecycle pluginlib tf2_geometry_msgs tf2_ros behaviortree_cpp_v3)
foreach(dep ${deps})
  find_package(${dep} REQUIRED)
endforeach()
rosidl_generate_interfaces(${PROJECT_NAME} "action/PathTranslate.action"
  DEPENDENCIES nav_msgs builtin_interfaces)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
add_library(rm_path_translate_behavior SHARED src/path_translate.cpp)
add_library(rm_path_translate_bt_node SHARED src/path_translate_bt.cpp)
foreach(target rm_path_translate_behavior rm_path_translate_bt_node)
  target_include_directories(${target} PRIVATE include)
  ament_target_dependencies(${target} ${deps})
  target_link_libraries(${target} "${typesupport}")
endforeach()
target_compile_definitions(rm_path_translate_bt_node PRIVATE BT_PLUGIN_EXPORT)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS rm_path_translate_behavior rm_path_translate_bt_node
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY behavior_trees DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()

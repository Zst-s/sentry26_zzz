cmake_minimum_required(VERSION 3.8)
project(cruise)
set(CMAKE_CXX_STANDARD 14)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(cruise_node src/cruise.cpp)
target_include_directories(cruise_node PRIVATE include)
ament_target_dependencies(cruise_node rclcpp geometry_msgs std_msgs tf2 tf2_ros)
install(TARGETS cruise_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/send_cruise_event.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(cruise_logic_test test/cruise_logic_test.cpp)
  target_include_directories(cruise_logic_test PRIVATE include)
endif()
ament_package()
